scripted_gui = {
	toolpack_open = {
		context_type = player_context
		window_name = "toolpack_open_container"
		visible = {
			is_ai = no
			NOT = { has_global_flag = toolpack_hidden }
			OR = {
				NOT = { has_global_flag = tp_is_multiplayer }
				has_global_flag = spm_everyone
				has_country_flag = tp_is_host
				NOT = { any_country = { has_country_flag = tp_is_host } }
			}
		}
		effects = {
			toolpack_open_click = {
				ROOT = { set_country_flag = spm_panel_enabled }
				if = {
					limit = { NOT = { has_variable = spm_div_count } }
					set_variable = { ROOT.spm_div_count = 1 }
					set_variable = { ROOT.spm_ship_count = 1 }
					set_variable = { ROOT.spm_ship_era = 1 }
				}
			}
			toolpack_close_at_right_click = {
				ROOT = { clr_country_flag = spm_panel_enabled }
				ROOT = { clr_country_flag = fxt_arm_restore }
			}
		}
		triggers = {
			toolpack_open_visible = {
				NOT = { ROOT = { has_country_flag = spm_panel_enabled } }
			}
			toolpack_close_at_right_visible = {
				ROOT = { has_country_flag = spm_panel_enabled }
			}
			spm_hint_visible = {
				ROOT = { has_country_flag = spm_panel_enabled }
			}
		}
	}
}
